   \]

   \[
   \frac{120 \text{ GB}}{2 \text{ GB/min}} = 60 \text{ minutes}
   \]

   \[
   20 \text{ minutes} + 60 \text{ minutes} = 80 \text{ minutes}
   \]

   \[
   80 \text{ minutes} + 20 \text{ minutes} = 100 \text{ minutes}
