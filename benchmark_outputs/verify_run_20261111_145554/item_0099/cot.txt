\[ \text{Initial cars} - \text{Cars driven through first 15 mins} = \text{Cars remaining after 15 mins} \]
\[ 30 - (\text{Cars driven through first 15 mins}) = 25 \]

\[ 30 - 25 = \text{Cars driven through first 15 mins} \]
\[ 5 = \text{Cars driven through first 15 mins} \]
